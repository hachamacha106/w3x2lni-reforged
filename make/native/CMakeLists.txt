cmake_minimum_required(VERSION 3.24)
project(W3x2lniNativeArchive LANGUAGES C CXX)
if(NOT DEFINED W2L_STORMLIB_SOURCE OR NOT DEFINED W2L_ZLIB_PREFIX)
    message(FATAL_ERROR "Pass verified StormLib source and staged zlib")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_BZip2 ON CACHE BOOL "Use pinned bundled copy" FORCE)
set(WITH_LIBTOMCRYPT OFF CACHE BOOL "Use pinned bundled copy" FORCE)
set(STORM_USE_BUNDLED_LIBRARIES OFF CACHE BOOL "Use verified external zlib" FORCE)
set(STORM_BUILD_TESTS OFF CACHE BOOL "Upstream tests need ALSA/game fixtures" FORCE)
set(STORM_SKIP_INSTALL ON CACHE BOOL "Builder stages verified artifacts" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "Build application DLL/test bridge" FORCE)
set(STORM_UNICODE ON CACHE BOOL "Match Windows wide-path FFI" FORCE)
set(ZLIB_USE_STATIC_LIBS ON)
set(ZLIB_INCLUDE_DIR "${W2L_ZLIB_PREFIX}/include" CACHE PATH "Verified headers" FORCE)
set(ZLIB_LIBRARY "${W2L_ZLIB_LIBRARY}" CACHE FILEPATH "Verified static zlib" FORCE)
set(ZLIB_LIBRARY_RELEASE "${W2L_ZLIB_LIBRARY}" CACHE FILEPATH "Verified static zlib" FORCE)
set(ZLIB_LIBRARY_DEBUG "${W2L_ZLIB_LIBRARY}" CACHE FILEPATH "Verified static zlib" FORCE)
find_package(ZLIB 1.3.2 EXACT REQUIRED)
file(REAL_PATH "${W2L_ZLIB_PREFIX}/include" expected_include)
file(REAL_PATH "${ZLIB_INCLUDE_DIR}" found_include)
file(REAL_PATH "${W2L_ZLIB_LIBRARY}" expected_library)
file(REAL_PATH "${ZLIB_LIBRARY_RELEASE}" found_library)
if(NOT found_include STREQUAL expected_include OR NOT found_library STREQUAL expected_library)
    message(FATAL_ERROR "zlib resolved outside verified prefix")
endif()
add_subdirectory("${W2L_STORMLIB_SOURCE}" stormlib)
get_target_property(storm_definitions storm COMPILE_DEFINITIONS)
if(NOT "__SYS_ZLIB" IN_LIST storm_definitions AND NOT "-D__SYS_ZLIB" IN_LIST storm_definitions)
    message(FATAL_ERROR "StormLib fell back to bundled zlib")
endif()
if("__SYS_BZLIB" IN_LIST storm_definitions OR "-D__SYS_BZLIB" IN_LIST storm_definitions)
    message(FATAL_ERROR "StormLib selected external BZip2")
endif()
get_target_property(storm_sources storm SOURCES)
get_target_property(storm_source_root storm SOURCE_DIR)
file(REAL_PATH "${W2L_STORMLIB_SOURCE}" expected_storm)
foreach(source IN LISTS storm_sources)
    if(source MATCHES "[$]<")
        message(FATAL_ERROR "Unexpected generated source")
    endif()
    file(REAL_PATH "${source}" source_path BASE_DIRECTORY "${storm_source_root}")
    cmake_path(IS_PREFIX expected_storm "${source_path}" NORMALIZE under_pin)
    if(NOT under_pin OR source_path MATCHES "/src/zlib/")
        message(FATAL_ERROR "Unpinned or bundled-zlib source: ${source_path}")
    endif()
endforeach()
get_target_property(storm_links storm LINK_LIBRARIES)
foreach(library IN LISTS storm_links)
    if(NOT library STREQUAL "ZLIB::ZLIB" AND NOT library STREQUAL "wininet")
        message(FATAL_ERROR "Unexpected StormLib dependency: ${library}")
    endif()
endforeach()
add_executable(native_probe probe.cpp)
target_include_directories(native_probe PRIVATE "${W2L_STORMLIB_SOURCE}/src")
target_link_libraries(native_probe PRIVATE storm ZLIB::ZLIB)
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "Retained application requires Windows x86")
    endif()
    target_compile_definitions(native_probe PRIVATE UNICODE _UNICODE)
    add_custom_command(TARGET native_probe POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "$<TARGET_FILE:storm>" "$<TARGET_FILE_DIR:native_probe>")
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/artifacts-$<CONFIG>.json" CONTENT
    "{\"library\":\"$<TARGET_FILE:storm>\",\"probe\":\"$<TARGET_FILE:native_probe>\",\"compiler\":\"${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\",\"c_compiler\":\"${CMAKE_C_COMPILER_ID} ${CMAKE_C_COMPILER_VERSION}\",\"static_zlib\":\"${W2L_ZLIB_LIBRARY}\"}\n")
